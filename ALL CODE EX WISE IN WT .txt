PROGRAM - JAVASCRIPT
function validateForm(){
let valid=true;
let name=document.getElementById("name").value.trim();
let email=document.getElementById("email").value.trim();
let phone=document.getElementById("phone").value.trim();
let education=document.getElementById("education").value;
let skills=document.getElementById("skills").value.trim();
let careerGoal=document.getElementById("careerGoal").value;
let experience=document.getElementById("experience").value;
let goals=document.getElementById("goals").value.trim();
document.getElementById("nameError").innerHTML="";
document.getElementById("emailError").innerHTML="";
document.getElementById("phoneError").innerHTML="";
document.getElementById("educationError").innerHTML="";
document.getElementById("skillsError").innerHTML="";
document.getElementById("careerGoalError").innerHTML="";
document.getElementById("experienceError").innerHTML="";
document.getElementById("goalsError").innerHTML="";
document.getElementById("successMessage").innerHTML="";
if(name===""){
document.getElementById("nameError").innerHTML="Please enter your full name.";
valid=false;
}
if(email===""){
document.getElementById("emailError").innerHTML="Please enter your email address.";
valid=false;
}else if(!email.includes("@")){
document.getElementById("emailError").innerHTML="Please enter a valid email address.";
valid=false;
}
if(phone===""){
document.getElementById("phoneError").innerHTML="Please enter your phone number.";
valid=false;
}else if(phone.length!==10){
document.getElementById("phoneError").innerHTML="Phone number must contain 10 digits.";
valid=false;
}
if(education===""){
document.getElementById("educationError").innerHTML="Please select your education.";
valid=false;
}
if(skills===""){
document.getElementById("skillsError").innerHTML="Please enter your skills.";
valid=false;
}
if(careerGoal===""){
document.getElementById("careerGoalError").innerHTML="Please select your career goal.";
valid=false;
}
if(experience===""){
document.getElementById("experienceError").innerHTML="Please select your experience.";
valid=false;
}
if(goals===""){
document.getElementById("goalsError").innerHTML="Please enter your career goals.";
valid=false;
}
if(valid){
document.getElementById("successMessage").innerHTML="Career Profile Created Successfully!";
}
return false;
}

EX.NO.4 - INSTALLATION OF APACHE TOMCAT WEBSERVER

PROCEDURE

STEP 1: Visit the Apache Tomcat Website
 Open a web browser and go to the official Apache Tomcat website. Download the appropriate Windows Installer (.exe) for the required Tomcat version.






STEP 2: Launch the Installer
Locate the downloaded Tomcat .exe file and open it. The Apache Tomcat Setup Wizard will be displayed. Click Next to proceed.


STEP 3: Accept the License Agreement
The license agreement will be shown on the screen. Review the terms and select I Agree to proceed with the installation. 

STEP 4: Select the Required Components
The installer will display the Choose Components screen. Select the necessary components for Tomcat and click Next.
 
STEP 5: Configure Tomcat
In the configuration screen, specify the HTTP connector port, usually 8080. Enter an administrator username and password for accessing Tomcat management applications, then click Next.


STEP 6: Proceed with the Installation
The installer will begin installing the required components. Allow the installation process to complete and click Close when the installation window indicates completion.

STEP 7: Specify the JDK Location
When the Java Virtual Machine (JVM) path is requested, browse to the location where the JDK is installed on the system. Select the appropriate JDK folder and click Next.

STEP 8: Finish the Setup
 After all installation files have been configured, the Apache Tomcat Setup Wizard Completed screen will appear. Click Finish to exit the installer.

STEP 9: Verify Tomcat Through Localhost
 Open a web browser and enter http://localhost:8080 in the address bar. If Tomcat is running correctly, the Apache Tomcat welcome page will be displayed.

STEP 10: Open the Management Application
 From the Tomcat welcome page, select Host Manager or Manager App. Provide the administrator credentials created during the configuration process to sign in.
 
STEP 11: Access Tomcat Web Application Manager
 After successful authentication, the Tomcat Web Application Manager will open. It provides options to deploy, start, stop, reload, and manage web applications.

PROGRAM - CAREER PROFILE.JAVA

public class CareerProfile {
    private String name,email,phone,education,skills,careerGoal,experience,goals;

    public CareerProfile(String name,String email,String phone,String education,String skills,String careerGoal,String experience,String goals){
        this.name=name;
        this.email=email;
        this.phone=phone;
        this.education=education;
        this.skills=skills;
        this.careerGoal=careerGoal;
        this.experience=experience;
        this.goals=goals;
    }
    public String getName(){return name;}
    public String getEmail(){return email;}
    public String getPhone(){return phone;}
    public String getEducation(){return education;}
    public String getSkills(){return skills;}
    public String getCareerGoal(){return careerGoal;}
    public String getExperience(){return experience;}
    public String getGoals(){return goals;}
}

import java.sql.*;
public class CareerProfileDAO {
    private String url="jdbc:sqlserver://103.207.1.91:1433;databaseName=CSE9596;encrypt=true;trustServerCertificate=true";
    private String username="MZ",password="mzcet@1234";
    public boolean saveProfile(CareerProfile p){
        String sql="INSERT INTO CareerProfiles(name,email,phone,education,skills,careerGoal,experience,goals) VALUES(?,?,?,?,?,?,?,?)";
        try{
            Class.forName("com.microsoft.sqlserver.jdbc.SQLServerDriver");
            Connection con=DriverManager.getConnection(url,username,password);
            PreparedStatement ps=con.prepareStatement(sql);
            ps.setString(1,p.getName());
            ps.setString(2,p.getEmail());
            ps.setString(3,p.getPhone());
            ps.setString(4,p.getEducation());
            ps.setString(5,p.getSkills());
            ps.setString(6,p.getCareerGoal());
            ps.setString(7,p.getExperience());
            ps.setString(8,p.getGoals());
            int result=ps.executeUpdate();

            ps.close();
            con.close();
            return result>0;
        }catch(Exception e){
            e.printStackTrace();
            return false;
        }
    }
}

PROGRAM - CAREER SERVLET.JAVA

import java.io.*;
import javax.servlet.*;
import javax.servlet.http.*;
public class CareerServlet extends HttpServlet {
    protected void doPost(HttpServletRequest request,HttpServletResponse response)throws ServletException,IOException{
        request.setCharacterEncoding("UTF-8");
        String name=request.getParameter("name");
        String email=request.getParameter("email");
        String phone=request.getParameter("phone");
        String education=request.getParameter("education");
        String skills=request.getParameter("skills");
        String careerGoal=request.getParameter("careerGoal");
        String experience=request.getParameter("experience");
        String goals=request.getParameter("goals");
        CareerProfile profile=new CareerProfile(name,email,phone,education,skills,careerGoal,experience,goals);
        CareerProfileDAO dao=new CareerProfileDAO();
        boolean success=dao.saveProfile(profile);
        if(success){
            request.setAttribute("name",name);
            request.setAttribute("email",email);
            request.setAttribute("phone",phone);
            request.setAttribute("education",education);
            request.setAttribute("skills",skills);
            request.setAttribute("careerGoal",careerGoal);
            request.setAttribute("experience",experience);
            request.setAttribute("goals",goals);
            RequestDispatcher rd=request.getRequestDispatcher("/profile-result.jsp");
            rd.forward(request,response);
        }else{
            response.setContentType("text/html;charset=UTF-8");
            PrintWriter out=response.getWriter();
            out.println("<h2>Career Profile Creation Failed</h2>");
            out.println("<p>Database insertion failed.</p>");
        }
    }
}
